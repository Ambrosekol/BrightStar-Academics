{#- Small helpers for writing documentation pages. Every reference they make is checked by tests/current/test_docs_contract.py, so a page can never point at a file, route or table that is gone. -#} {# A source file, linked to its entry in the module reference. #} {% macro file(path) -%} {{ path }} {%- endmacro %} {# A page template, linked to the template reference. #} {% macro tpl(name) -%} templates/{{ name }} {%- endmacro %} {# A route, by its endpoint name, linked to the route reference. #} {% macro route(endpoint) -%} {{ endpoint }} {%- endmacro %} {# A database table, linked to the data model. #} {% macro table(name) -%} {{ name }} {%- endmacro %} {# A permission key, linked to the permission reference. #} {% macro perm(key) -%} {{ key }} {%- endmacro %} {# A page of these docs. #} {% macro page(slug, label) -%} {{ label }} {%- endmacro %}
